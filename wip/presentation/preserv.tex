% -------------------------------------------------------------------
% Frame 1: Core Definitions & Target Theorem Statement
% -------------------------------------------------------------------
\begin{frame}[t]{Some definitions}
\begin{align*}
	Subst &:= set \ (Var * Exp) \\
	Problem &:= set \ Equation \\
	Tuple &:= set \ (Subst * Problem) \\
	\_\langle \_\rangle &: exp \to Subst \to exp \\
	dom &: Subst \to set \ Var \\
	smatch &: Tuple \to Tuple \to Prop
\end{align*}

\end{frame}

% -------------------------------------------------------------------
% Frame 2: Side-by-Side Proof Walkthrough (Definitions Left, Steps Right)
% -------------------------------------------------------------------
\begin{frame}[t]{\textit{smatch} steps are classified into three}
\begin{block}{Congruence steps}
	\begin{align*}
		smatch\_app\_congr : \ \  &(s \ t =_? s' \ t') \in P \to \\
		                    & smatch \ (S,P) \ (S, P + (s =_? s') + (t=_?t') \ \backslash \ (s  t =_? s' \ t'))  
	\end{align*}
\end{block}
\pause
\begin{block}{$\sigma$-min steps}
	\begin{align*}
		smatch\_subst\_comp : \ \ &(s[\sigma \circ \tau] =_? s'[\rho]) \in P \to \\
		                        & smatch \ (S,P) \ (S, P + (s[\sigma][\tau] =_? s'[\rho]) \backslash (s[\sigma \circ \tau]=_?s'[\rho])) 
	\end{align*}
\end{block}
\pause
\begin{block}{Instantiation case}
	\begin{align*}
		smatch\_inst : \ \ & (s=_?X) \in P \to \\
		                   & X \notin lhs\_vars \ P \to \\
		                   & S' =_{\sigma_{min}} S \gg \{(X,s)\} \to \\
		                    &  smatch \ (S,P) \ (S', (P \backslash (s=_?X))\langle \{(X,s)\} \rangle)  
	\end{align*}
\end{block}
\end{frame}

\begin{frame}[t]{Preservation statement}
\begin{block}{Validity}
	\begin{align*}
		valid\_tuple \ (S,P) :=  \underline{(dom \ S) \cap (vars\_of \ P) = \phi}
	\end{align*}
\end{block}

\begin{block}{Matching solution}
	\begin{align*}
		match\_sol \ Sl \ (S,P) \ := \ &\underline{(s=_?t) \in P \to s =_{\sigma_{min}} t\langle Sl\rangle} \ \wedge  \\
		&\underline{\exists S', S \gg S' =_{\sigma_{min}} Sl} 
	\end{align*}
\end{block}

\begin{block}{Preservation theorem}
	\begin{align*}
	   match\_pres  \ : \ &valid\_tuple \ T \to \\
	                     &(dom \ Sl) \cap (lhvars\_of \ P) \to \\
	                     & smatch \ T \ T' \to  \\
	                     & match\_sol \ Sl \ T' \to \\
	                     & match\_sol \ Sl \ T 
	\end{align*}
\end{block}
\end{frame}



\begin{frame}[fragile]{Proof proceeds by case analysis on $smatch$}
	\begin{block}{Inst case }
		\begin{lstlisting}[language=Coq, escapeinside={(*}{*)}]
		    (*$H_1$*) : (*$(dom \ S) \ \cap \ vars\_of \ P = \{\}$*)
		    (*$H_2$*) : (*$(dom \ Sl) \cap \ lhvars\_of \ P = \{\}$*)
		    (*$H_3$*) : (*$X \notin lhvars\_of \ P$*)
		    (*$H_4$*) : (*$match\_sol \ Sl \ (S \gg \{(X,s)\}, \ (P \ \backslash \ (s,X))\langle \{(X,s)\} \rangle$*) 
			----------------------------------
			(*$match\_sol \ Sl \ P$*)
		\end{lstlisting}
	\end{block}
\end{frame}
